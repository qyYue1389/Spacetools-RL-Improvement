#

# not the env's. install_graspgen.sh runs `conda install cuda-toolkit=12.1` but the env

# major 12 -> 13 makes torch raise. 12.8 shares torch's major, so it only warns.
# Host compiler must be gcc <= 14 (system gcc-13); conda injects
# NVCC_PREPEND_FLAGS=-ccbin=<conda gcc 15.3>, which must be overridden.
